// ===== 초음파 센서 거리 측정 =====
// Trig = 2번 핀 (초음파 발사)
// Echo = 3번 핀 (반사된 초음파 수신)

void setup() {
  pinMode(2, OUTPUT); // Trig = 출력
  pinMode(3, INPUT);  // Echo = 입력
  Serial.begin(9600); // PC 시리얼 통신 시작
}

void loop() {
  // 1. 초음파 발사
  digitalWrite(2, LOW);  // Trig 초기화
  delayMicroseconds(2);  // 2마이크로초 대기
  digitalWrite(2, HIGH); // 초음파 발사 신호 시작
  delayMicroseconds(10); // 10마이크로초 동안 HIGH 유지
  digitalWrite(2, LOW);  // 발사 신호 종료

  // 2. 초음파 왕복 시간 측정
  long duration = pulseIn(3, HIGH);
  // pulseIn(핀번호, HIGH) = HIGH 펄스 유지시간 측정(μs)
  // long = 큰 정수를 저장하는 자료형

  if (duration == 0) {
    return; // 초음파 응답이 없으면 이번 loop 종료
  }

  // 3. 거리 계산
  long distance = duration / 58.2;
  // 왕복시간(μs)을 거리(cm)로 변환
  // 초음파 1cm 이동에 약 29.1μs
  // 왕복이므로 29.1 × 2 = 58.2로 나눔

  // 4. 시리얼 모니터 출력
  char pr[24]; // 출력 문자열 저장 공간
  sprintf(pr, "distance = %ld cm", distance);
  // sprintf(저장변수, "출력형식", 입력값)
  // %ld = long 정수 출력 형식

  Serial.println(pr); // PC에 거리 출력
  delay(200);         // 0.2초마다 반복
}


// ===== 시험용 함수 정리 =====
// pinMode(핀, OUTPUT) = 출력 설정 (Trig)
// pinMode(핀, INPUT)  = 입력 설정 (Echo)
// digitalWrite(핀, HIGH) = HIGH 출력
// digitalWrite(핀, LOW)  = LOW 출력
// delayMicroseconds(10) = 0.00001초 대기
// pulseIn(3, HIGH) = HIGH 펄스의 길이 측정(μs)
// if(duration == 0) return; = 측정 실패 시 이번 반복 종료
// distance = duration / 58.2 = 거리(cm) 계산
// Serial.begin(9600) = 통신속도 설정
// Serial.println(pr) = 시리얼 모니터 출력
// delay(200) = 0.2초 대기

// Trig = 초음파 발사 / Echo = 반사 신호 수신

// ===== 서보모터 함수 정리 =====
// #include <Servo.h> = 서보모터 라이브러리 불러오기
// Servo myservo; = 서보모터 객체(변수) 선언
// myservo.attach(핀번호) = 서보모터 연결 핀 지정
// myservo.write(각도) = 서보모터 회전 (보통 0~180도)
// delay(시간) = 모터가 움직일 시간 확보 (ms)

#include <Servo.h>
Servo myservo;

void setup() {
  myservo.attach(9); // 서보모터를 9번 핀에 연결
}

void loop() {

  // 1. 특정 각도로 회전
  myservo.write(90); // 90도로 이동
  delay(1000);       // 이동할 시간 확보 + 1초 대기


  // 2. 두 각도 반복 이동
  myservo.write(120); // 120도로 이동
  delay(1000);        // 1초 대기
  myservo.write(0);   // 0도로 이동
  delay(1000);        // 1초 대기


  // 3. 서서히 회전하기
  for (int i = 0; i < 120; i++) {
    // i=0 시작 / i<120 = 119까지 / i++ = 1씩 증가
    myservo.write(i); // 0도부터 119도까지 1도씩 회전
    delay(15);        // 각도 변경 후 0.015초 대기
                       // 모터가 움직일 시간 확보
  }

  myservo.write(0); // 다시 0도로 이동
  delay(1000);      // 1초 대기 후 반복
}

// ===== 시험 수정 포인트 =====
// attach(9) = 서보모터 연결 핀 번호
// write(90) = 회전할 목표 각도
// i<120 = 0~119도 반복 (120도 포함은 i<=120)
// i++ = 1도씩 증가 / i-- = 1도씩 감소
// delay(15) = 1도 이동 명령마다 0.015초 대기
// delay(1000) = 1초 대기
// write()는 목표 각도를 명령하고 바로 다음 코드로 넘어감
// 따라서 모터가 이동할 시간을 delay()로 확보
