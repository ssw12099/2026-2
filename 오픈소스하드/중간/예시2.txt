//무드등 5
//삼색 b9 g10 r11
//트리그 7 에코 8
//버튼 4

// ---음계---
// [4옥타브]
// 도(C)  = 262Hz
// 레(D)  = 294Hz
// 미(E)  = 330Hz
// 파(F)  = 349Hz
// 솔(G)  = 392Hz
// 라(A)  = 440Hz
// 시(B)  = 494Hz


#include <Servo.h>
Servo mys;

int seroc = 0;
int mode = 0;  //0 auto # 1 man
int change = 0;

int au[3] = { 262, 330, 392 };
int ma[3] = { 392, 330, 262 };

void setup() {
  // put your setup code here, to run once:
  Serial.begin(9600);

  pinMode(5, OUTPUT);
  pinMode(9, OUTPUT);
  pinMode(10, OUTPUT);
  pinMode(11, OUTPUT);
  pinMode(2, OUTPUT);
  pinMode(4, INPUT);
  pinMode(7, OUTPUT);
  pinMode(8, INPUT);

  mys.attach(13);
  mys.write(0);
}

void loop() {
  // put your main code here, to run repeatedly:
  if (digitalRead(4) == HIGH) {
    mode = (mode == 0) ? 1 : 0;
    change = 1;
  }
  digitalWrite(7, LOW);
  delayMicroseconds(2);
  digitalWrite(7, HIGH);
  delayMicroseconds(10);
  digitalWrite(7, LOW);
  long dura = pulseIn(8, HIGH);
  long dis = dura / 58.2;

  int jodo = analogRead(A0);

  char pr[50];
  sprintf(pr, "LIGHT = %d, Dist = %ld, Mode = %s", jodo, dis, (mode == 0) ? "AUTO" : "MANUAL");
  Serial.println(pr);

  if (dis <= 30) {
    analogWrite(5, map(jodo, 0, 1023, 255, 0));
  }else{
    analogWrite(5,0);
  }

  if (mode == 1) {
    if(change == 1){
      for(int i=0;i<3;i++){
      tone(2,au[i],200);
      delay(220);
      }
      change = 0;
    }

    analogWrite(9, 255);
    analogWrite(10, 0);
    analogWrite(11, 255);
  } else {
    if(change == 1){
      for(int i=0;i<3;i++){
      tone(2,ma[i],200);
      delay(220);
    }
      change = 0;
    }
  

    if (jodo < 300) {
      analogWrite(9, 255);
      analogWrite(10, 0);
      analogWrite(11, 0);
      if (seroc != 0) {
        mys.write(0);
        seroc = 0;
        tone(2, 262, 200);
      }
    } else if (jodo < 700) {
      analogWrite(9, 0);
      analogWrite(10, 255);
      analogWrite(11, 0);
      if (seroc != 0 && jodo < 600) {
        mys.write(0);
        seroc = 0;
        tone(2, 262, 200);
      }
    } else {
      analogWrite(9, 0);
      analogWrite(10, 0);
      analogWrite(11, 255);
      if (seroc != 1) {
        mys.write(90);
        seroc = 1;
        tone(2, 523, 200);
      }
    }
  }

  delay(200);
}
