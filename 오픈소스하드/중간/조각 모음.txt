   WEEK 2 - LED 제어

// [변수 선언] LED가 연결된 핀 번호 지정
int Rl = 13;  // 빨간 LED = 13번 핀
int Yl = 10;  // 노란 LED = 10번 핀
int Gl = 9;   // 초록 LED = 9번 핀
// 핀 번호가 바뀌면 여기 숫자만 수정


void setup() {
  // [setup] 프로그램 시작 시 딱 1번 실행

  // [pinMode] 핀을 입력 또는 출력으로 설정
  // 형식: pinMode(핀번호, OUTPUT);
  // OUTPUT = 출력 / INPUT = 입력

  pinMode(13, OUTPUT); // 13번 핀을 출력으로 설정
  pinMode(Rl, OUTPUT); // Rl(13번)을 출력으로 설정
  pinMode(Yl, OUTPUT); // Yl(10번)을 출력으로 설정
  pinMode(Gl, OUTPUT); // Gl(9번)을 출력으로 설정

  // pinMode(13,OUTPUT)과 pinMode(Rl,OUTPUT)은 같은 뜻
  // 둘 중 하나만 사용해도 됨
}


void loop() {
  // [loop] 프로그램이 끝없이 반복 실행됨


  // ==============================
  // 1. LED 서서히 밝게 켜기
  // ==============================

  for (int i = 0; i < 256; i++) {
    // for(초기값; 반복조건; 증감)
    // i=0     : 0부터 시작
    // i<256   : 255까지 반복
    // i++     : 1씩 증가
    // 0=꺼짐 / 255=최대 밝기

    analogWrite(Gl, i);
    // 형식: analogWrite(핀번호, PWM값);
    // Gl(9번) LED의 밝기를 i값으로 조절
    // PWM 출력 범위: 0~255
    // PWM 지원 디지털 핀에서 사용

    delay(10);
    // 10ms = 0.01초 대기
    // 숫자가 클수록 밝아지는 속도가 느려짐
  }


  // ==============================
  // 2. LED 서서히 어둡게 끄기
  // ==============================

  for (int i = 255; i >= 0; i--) {
    // i=255  : 최대 밝기부터 시작
    // i>=0   : 0까지 반복
    // i--    : 1씩 감소
    // 255에서 0으로 내려가면서 어두워짐

    analogWrite(Gl, i); // LED 밝기 감소
    delay(10);          // 감소 속도 조절
  }


  // ==============================
  // 3. LED 밝기 직접 지정
  // ==============================

  // analogWrite(Gl, 0);   // LED 끄기
  // analogWrite(Gl, 128); // LED 중간 밝기
  // analogWrite(Gl, 255); // LED 최대 밝기

  // 숫자만 바꾸면 밝기 조절 가능
  // analogWrite는 실제 아날로그 전압이 아닌 PWM 출력


  // ==============================
  // 4. 디지털 신호로 LED 켜기/끄기
  // ==============================

  digitalWrite(Rl, HIGH);
  // 형식: digitalWrite(핀번호, HIGH 또는 LOW);
  // HIGH = 1 = 켜기 (UNO 기준 약 5V)
  // LOW  = 0 = 끄기 (약 0V)
  // HIGH(5V)처럼 전압값을 직접 넣지 않음

  delay(1000); // 켠 상태로 1초 유지

  digitalWrite(Rl, LOW); // LED 끄기


  // ==============================
  // 5. delay 시간 조절
  // ==============================

  // delay(1);     // 0.001초
  // delay(10);    // 0.01초
  // delay(100);   // 0.1초
  // delay(500);   // 0.5초
  // delay(1000);  // 1초
  // delay(2000);  // 2초

  delay(1000); // 1초 기다린 후 loop 다시 시작
}
